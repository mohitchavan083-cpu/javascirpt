Variable (var), let & const

console.log("----- push() -----") let fruits = ['apple', 'banana']; console.log(fruits); fruits.push('orange', 'grape'); console.log(fruits); console.log("----- pop() -----") let fruit = ['apple', 'banana', 'orange']; console.log(fruit); let lastFruit = fruit.pop(); console.log(lastFruit); console.log(fruit); console.log("----- unshift() -----") // adds elements to the start let fruit_unshift = ['banana', 'orange']; console.log(fruit_unshift); fruit_unshift.unshift('apple', 'grape'); console.log(fruit_unshift); console.log("----- shift() -----") // remove elements from the start let fruit_shift = ['apple', 'banana', 'orange']; console.log(fruit_shift); let firstFruit = fruit_shift.shift(); console.log(firstFruit); console.log(fruit_shift); console.log("----- length property -----") let fruit_length = ['apple', 'banana', 'mango', 'orange']; console.log(fruit_length); fruit_length.length = 2; // truncating the array console.log(fruit_length); console.log("===== shift() =====") // remove elements from the start let fruit_shift = ['apple', 'banana', 'orange'];
